# ============================================================================
# ExAct 全项目依赖 —— OpenVLA 版（单文件安装，与 requirements.txt 二选一）
# ============================================================================
# 背景（2026-08-30 决定）：
#   - lerobot 只用于 VLA 后端，与 OpenVLA 硬冲突（transformers==4.40.1 vs
#     lerobot[smolvla] 的 >=5.4 / torch==2.2.0 vs >=2.7，交集为空）。
#     决定：不再安装 lerobot，直接以 OpenVLA 作为唯一 VLA 后端。
#   - 主系统其余依赖（仿真/图像/配置/Agent 编排/测试）与 OpenVLA 依赖无冲突，
#     因此【一个文件、一个环境】即可装齐整个项目，无需先装 requirements.txt。
#   - 本文件 = OpenVLA 官方锁定依赖 + 主系统全部依赖（不含 lerobot）。
#   - 原 requirements.txt 是 lerobot 路线，与本文互斥，二选一。
#
# 【安装（一条命令）】
#   conda create -n openvla python=3.11 -y && conda activate openvla   # Python 3.10~3.12 均可
#   pip install -r requirements-openvla.txt
#   pip check
#
# 【环境要求】
#   - Python 3.10~3.12 均可：OpenVLA 官方锁 torch 2.2.0，官方发布了 cp312 wheel
#     （3.12 可直接 pip 安装）；pybullet 未锁版本，pip 自动选当前 Python 的预编译 wheel。
#   - 注意：sentencepiece 需 >=0.2.0 才有 cp312 wheel（0.1.99 在 3.12 下会触发源码编译）。
#
# 【torch 平台说明】
#   - torch==2.2.0 声明在下方，pip 按平台自动选 wheel：
#       Linux + NVIDIA GPU → PyPI 默认 CUDA 11.8 版（自动带 nvidia-* 依赖）
#       macOS / 无 GPU     → pip 自动选 CPU wheel
#   - 想改用 conda 装 CUDA 版（则删除下方 torch 三行，避免 pip 覆盖）：
#       conda install pytorch==2.2.0 torchvision==0.17.0 torchaudio==2.2.0 \
#           pytorch-cuda=12.1 -c pytorch -c nvidia
#   - bitsandbytes（4bit 量化）仅 Linux+CUDA 可用；macOS 装不上可注释掉该行。
# ============================================================================

# --- PyTorch 2.2（OpenVLA 官方锁定；torch 2.2.0 官方提供 cp312 wheel，Python 3.10~3.12 可用）---
torch==2.2.0

torchaudio==2.2.0

# --- OpenVLA 核心（官方 pyproject == 锁定，勿放宽）---

tokenizers==0.19.1          # transformers 4.40.1 配套约束 <0.20,>=0.19

sentencepiece>=0.2.0       # Llama tokenizer 加载；>=0.2.0 才提供 cp312 wheel（3.12 可用）
accelerate>=0.25.0          # low_cpu_mem_usage=True 低内存加载
einops                      # Prismatic 视觉骨干
huggingface_hub<1.0         # transformers 4.40.1 要求 <1.0,>=0.19.3
# flash-attn==2.5.5         # 可选加速（需编译）；缺包时 openvla_vla.py 自动回退 eager

pybullet

                                    # 新版 opencv 收紧到 numpy>=2，避免冲突
