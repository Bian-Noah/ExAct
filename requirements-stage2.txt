# ⚠️  安装约定（写入此文件，防止 pip 覆盖 conda 提供的 CUDA 版 PyTorch）
# ----------------------------------------------------------------------------
#  1. PyTorch CUDA 由 conda 提供（pytorch 2.5.1 + cuda12.4 + cudnn9.1.0），
#     本文件【不】包含 torch/torchvision/torchaudio，禁止通过 pip 装这三个包。
#  2. pip install 永远加 --no-deps，避免 pip 拉传递依赖时偷偷重装 torch。
#  3. 优先 conda install -c conda-forge；conda-forge 没有的（如 lerobot）才
#     走 pip --no-deps。
#  4. ⚠️  --no-deps 只允许用在 lerobot 自身！其他所有包都走默认 pip install，
#       让 pip 自动解决传递依赖。原因：--no-deps 会人为跳过 draccus → typing-inspect
#       → mypy_extensions 这条链，导致每个包"装上但用不了"。只有 lerobot 的
#       Requires-Dist 含 torch/torchvision，需要手动保护避免 pip 覆盖 conda 的 CUDA 版。
#  5. 装完跑 pip check 验证依赖闭环；缺啥补啥（用普通 pip install，不要 --no-deps）。
#
# 标准安装流程（在 robot conda env 内执行）：
#   conda activate robot
#   # 1) 与 torch 配套的 CUDA torchvision（必须 conda，否则覆盖 conda 的 torch）
#   conda install -c pytorch torchvision==0.20.1
#   # 2) conda-forge 上的运行时工具（hf/accelerate/bnb 与 conda 的 CUDA torch 配套最稳）
#   conda install -c conda-forge accelerate bitsandbytes tokenizers
#   # 3) 其他小依赖走 pip，默认安装（不要 --no-deps，让 pip 自动拉传递依赖）
#   pip install draccus opencv-python-headless termcolor
#   # 4) lerobot 本身是唯一允许 --no-deps 的包（防 pip 抢走 conda 的 CUDA torch）
#   pip install --no-deps lerobot
#   # 5) 闭环校验
#   pip check
#
# ❌ 禁止用法：
#   pip install -r requirements-stage2.txt          # 会从 PyPI 拉 CPU 版 torch
#   pip install torch torchvision torchaudio        # 会覆盖 conda 的 CUDA 版
# ============================================================================
